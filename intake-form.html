// ============================================================ // ALBERT HIBBERT ATTORNEYS - FICA INTAKE CONNECTION LAYER // Google Apps Script . Paste this into script.google.com // Version 1.0 . May 2026 // ============================================================ // -- CONFIGURATION ------------------------------------------- // After pasting this script, update the values below once. const CONFIG = { SHEET_NAME: "FICA Clients", // Name of your Google Sheet tab NOTIFY_EMAIL: "aha3@hibbertlaw.co.za", // Your email for notifications NOTIFY_NAME: "Mikhael Venter", // Your name FIRM_NAME: "Albert Hibbert Attorneys Inc.", DRIVE_FOLDER_NAME: "FICA Client Documents", // Google Drive folder name LINK_EXPIRY_HOURS: 72, // How long intake links stay active WHATSAPP_ENABLED: false, // Set true when Phase 2 is ready }; // -- SHEET COLUMNS -------------------------------------------- // These are the column headers created automatically on first run. const COLUMNS = [ "Timestamp", "Reference", "Client Name", "Client Type", "Email", "Phone", "ID / Reg Number", "Address", "Occupation / Representative", "Beneficial Owners", "Documents Uploaded", "Document Folder Link", "Status", "Verified By", "Verified Date", "Notes", "Link Token", "Link Expiry", "Link Used", ]; // ============================================================ // 1. SETUP - Run this once to initialise the spreadsheet // ============================================================ function setupSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(CONFIG.SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(CONFIG.SHEET_NAME); } // Write headers if the sheet is empty if (sheet.getLastRow() === 0) { sheet.appendRow(COLUMNS); sheet.getRange(1, 1, 1, COLUMNS.length) .setFontWeight("bold") .setBackground("#F1EFE8"); sheet.setFrozenRows(1); sheet.setColumnWidth(1, 160); sheet.setColumnWidth(3, 200); sheet.setColumnWidth(5, 200); sheet.setColumnWidth(12, 260); } // Ensure the Drive folder exists getOrCreateFolder(CONFIG.DRIVE_FOLDER_NAME); Logger.log("Setup complete. Sheet and Drive folder are ready."); SpreadsheetApp.getUi().alert( "Setup complete!\n\nSheet '" + CONFIG.SHEET_NAME + "' is ready and your Drive folder '" + CONFIG.DRIVE_FOLDER_NAME + "' has been created.\n\nNext step: deploy this script as a Web App." ); } // ============================================================ // 2. WEB APP ENTRY POINT - handles form submissions (POST) // and link validation (GET) // ============================================================ function doGet(e) { const token = e.parameter.token; if (!token) return jsonResponse({ valid: false, reason: "No token supplied" }); const result = validateToken(token); return jsonResponse(result); } function doPost(e) { try { var data; if (e.parameter && e.parameter.payload) { data = JSON.parse(e.parameter.payload); } else if (e.postData && e.postData.contents) { data = JSON.parse(e.postData.contents); } if (!data) return jsonResponse({ success: false, error: "No data received" }); var action = data.action; if (action === "generate_link") return jsonResponse(generateClientLink(data)); if (action === "submit_fica") return jsonResponse(handleFicaSubmission(data)); if (action === "activate") return jsonResponse(activateClient(data)); return jsonResponse({ success: false, error: "Unknown action" }); } catch (err) { return jsonResponse({ success: false, error: err.message }); } } // ============================================================ // 3. GENERATE INTAKE LINK - called from your dashboard // when you click "Generate link" for a new client // ============================================================ function generateClientLink(data) { const sheet = getSheet(); const token = generateToken(); const expiry = new Date(); expiry.setHours(expiry.getHours() + CONFIG.LINK_EXPIRY_HOURS); const ref = data.ref || generateRef(); const row = [ new Date(), // Timestamp ref, // Reference data.clientName, // Client Name data.clientType, // Client Type data.email, // Email String(data.phone || ""), // Phone "", // ID / Reg Number (filled on submission) "", // Address "", // Occupation "", // Beneficial Owners "", // Documents Uploaded "", // Document Folder Link "Link Sent", // Status "", // Verified By "", // Verified Date "", // Notes token, // Link Token expiry.toISOString(),// Link Expiry "false", // Link Used ]; sheet.appendRow(row); // Build the intake URL - points to the hosted intake form on GitHub const intakeUrl = "https://mikhael1999-beep.github.io/fica-intake/intake-form.html?token=" + token; // Send the link to the client by email sendLinkEmail(data.email, data.clientName, intakeUrl, ref, expiry); Logger.log("Link generated for " + data.clientName + " - " + ref); return { success: true, ref, token, intakeUrl, expiry: expiry.toISOString() }; } // ============================================================ // 4. HANDLE FICA SUBMISSION - fires when client submits // their intake form // ============================================================ function handleFicaSubmission(data) { const sheet = getSheet(); const token = data.token || ""; // Walk-in / reception mode — token starts with WALK_IN_ const isWalkIn = token.indexOf("WALK_IN_") === 0; if (isWalkIn) { // Create a new record directly from submitted data const ref = generateRef(); const clientName = data.entityName || (data.firstName + " " + data.lastName).trim() || "Walk-in Client"; const row = [ new Date(), ref, clientName, data.clientType || "Individual", data.email || "", data.phone || "", data.idNumber || data.regNumber || "", data.address || data.regAddress || "", data.occupation || data.repTitle || "", data.beneficialOwners || "", (data.documents || []).join(", "), "", "Pending Review", "", "", "", token, "", "true" ]; sheet.appendRow(row); const rowIndex = sheet.getLastRow(); sheet.getRange(rowIndex, 1, 1, COLUMNS.length).setBackground("#FAEEDA"); const parentFolder = getOrCreateFolder(CONFIG.DRIVE_FOLDER_NAME); const clientFolder = parentFolder.createFolder(ref + " - " + clientName); const folderUrl = clientFolder.getUrl(); const colMap = {}; COLUMNS.forEach((c, i) => colMap[c] = i + 1); sheet.getRange(rowIndex, colMap["Document Folder Link"]).setValue(folderUrl); sendSubmissionNotification(clientName, ref, data, folderUrl); return { success: true, ref, folderUrl }; } // Find the row for this token const rowIndex = findRowByToken(token); if (rowIndex === -1) return { success: false, error: "Invalid or expired token" }; const row = sheet.getRange(rowIndex, 1, 1, COLUMNS.length).getValues()[0]; // Check expiry const expiry = new Date(row[COLUMNS.indexOf("Link Expiry")]); if (new Date() > expiry) return { success: false, error: "This link has expired" }; // Check if already used if (row[COLUMNS.indexOf("Link Used")] === "true") { return { success: false, error: "This link has already been used" }; } // Create a client folder in Google Drive const clientName = row[COLUMNS.indexOf("Client Name")]; const ref = row[COLUMNS.indexOf("Reference")]; const folderName = ref + " - " + clientName; const parentFolder = getOrCreateFolder(CONFIG.DRIVE_FOLDER_NAME); const clientFolder = parentFolder.createFolder(folderName); const folderUrl = clientFolder.getUrl(); // Update the sheet row with submitted data const colMap = {}; COLUMNS.forEach((c, i) => colMap[c] = i + 1); sheet.getRange(rowIndex, colMap["ID / Reg Number"]).setValue(data.idNumber || ""); sheet.getRange(rowIndex, colMap["Address"]) .setValue(data.address || ""); sheet.getRange(rowIndex, colMap["Occupation / Representative"]).setValue(data.occupation || ""); sheet.getRange(rowIndex, colMap["Beneficial Owners"]).setValue(data.beneficialOwners || ""); sheet.getRange(rowIndex, colMap["Documents Uploaded"]).setValue( (data.documents || []).join(", ") ); sheet.getRange(rowIndex, colMap["Document Folder Link"]).setValue(folderUrl); sheet.getRange(rowIndex, colMap["Status"]) .setValue("Pending Review"); sheet.getRange(rowIndex, colMap["Link Used"]) .setValue("true"); // Colour-code the row as pending sheet.getRange(rowIndex, 1, 1, COLUMNS.length).setBackground("#FAEEDA"); // Notify the attorney sendSubmissionNotification(clientName, ref, data, folderUrl); Logger.log("FICA submission received: " + ref + " - " + clientName); return { success: true, ref, folderUrl }; } // ============================================================ // 5. ACTIVATE CLIENT - called from your dashboard when you // click "Activate client" after manual verification // ============================================================ function activateClient(data) { const sheet = getSheet(); const rowIndex = findRowByRef(data.ref); if (rowIndex === -1) return { success: false, error: "Reference not found" }; const colMap = {}; COLUMNS.forEach((c, i) => colMap[c] = i + 1); sheet.getRange(rowIndex, colMap["Status"]) .setValue("Active"); sheet.getRange(rowIndex, colMap["Verified By"]) .setValue(data.verifiedBy || CONFIG.NOTIFY_NAME); sheet.getRange(rowIndex, colMap["Verified Date"]).setValue(new Date().toLocaleDateString("en-ZA")); if (data.notes) { sheet.getRange(rowIndex, colMap["Notes"]).setValue(data.notes); } // Colour-code as active sheet.getRange(rowIndex, 1, 1, COLUMNS.length).setBackground("#EAF3DE"); // Send activation confirmation to client const email = sheet.getRange(rowIndex, colMap["Email"]).getValue(); const name = sheet.getRange(rowIndex, colMap["Client Name"]).getValue(); sendActivationEmail(email, name, data.ref); Logger.log("Client activated: " + data.ref); return { success: true }; } // ============================================================ // 6. EMAIL TEMPLATES // ============================================================ function sendLinkEmail(email, name, url, ref, expiry) { const expiryStr = expiry.toLocaleDateString("en-ZA", { day: "numeric", month: "long", year: "numeric", hour: "2-digit", minute: "2-digit" }); const subject = CONFIG.FIRM_NAME + " - FICA onboarding . " + ref; const body = `Dear ${name},\n\n` + `${CONFIG.FIRM_NAME} requires that all clients complete a FICA (Financial Intelligence Centre Act) onboarding process before we may act on your behalf.\n\n` + `Please complete your FICA application using the secure link below:\n\n` + ` ${url}\n\n` + `This link is unique to you and will expire on ${expiryStr}. It may only be used once.\n\n` + `You will need to have the following ready:\n` + ` . A certified copy of your identity document or company registration certificate\n` + ` . Proof of address not older than 3 months\n` + ` . Any other supporting documents applicable to your client type\n\n` + `Should you experience any difficulty, please contact our offices.\n\n` + `Kind regards,\n${CONFIG.NOTIFY_NAME}\n${CONFIG.FIRM_NAME}`; GmailApp.sendEmail(email, subject, body); Logger.log("Intake link email sent to " + email); } function sendSubmissionNotification(clientName, ref, data, folderUrl) { const subject = "FICA submission received - " + ref + " - " + clientName; const body = `A new FICA submission has been received and is awaiting your review.\n\n` + `Client: ${clientName}\n` + `Reference: ${ref}\n` + `Type: ${data.clientType || "-"}\n` + `Email: ${data.email || "-"}\n` + `Phone: ${data.phone || "-"}\n\n` + `Documents have been placed in the client's Drive folder:\n${folderUrl}\n\n` + `Log in to the FICA dashboard to review and activate this client.\n\n` + `Kind regards,\n${CONFIG.FIRM_NAME} FICA System`; GmailApp.sendEmail(CONFIG.NOTIFY_EMAIL, subject, body); } function sendActivationEmail(email, name, ref) { const subject = CONFIG.FIRM_NAME + " - FICA approved . " + ref; const body = `Dear ${name},\n\n` + `We are pleased to confirm that your FICA application (reference ${ref}) has been ` + `reviewed and approved. Your client status is now ACTIVE.\n\n` + `${CONFIG.FIRM_NAME} is now able to act on your behalf in terms of the relevant mandate.\n\n` + `Should you have any queries, please do not hesitate to contact our offices.\n\n` + `Kind regards,\n${CONFIG.NOTIFY_NAME}\n${CONFIG.FIRM_NAME}`; GmailApp.sendEmail(email, subject, body); Logger.log("Activation email sent to " + email); } // ============================================================ // 7. UTILITY FUNCTIONS // ============================================================ function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); return ss.getSheetByName(CONFIG.SHEET_NAME); } function getOrCreateFolder(name) { const folders = DriveApp.getFoldersByName(name); if (folders.hasNext()) return folders.next(); return DriveApp.createFolder(name); } function generateToken() { const chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789"; let token = ""; for (let i = 0; i < 32; i++) { token += chars.charAt(Math.floor(Math.random() * chars.length)); } return token; } function generateRef() { const year = new Date().getFullYear(); const num = String(Math.floor(Math.random() * 900) + 100); return "AH/" + year + "/" + num; } function findRowByToken(token) { const sheet = getSheet(); if (!sheet || sheet.getLastRow() < 2) return -1; const col = COLUMNS.indexOf("Link Token") + 1; const data = sheet.getRange(2, col, sheet.getLastRow() - 1, 1).getValues(); for (let i = 0; i < data.length; i++) { if (data[i][0] === token) return i + 2; } return -1; } function findRowByRef(ref) { const sheet = getSheet(); if (!sheet || sheet.getLastRow() < 2) return -1; const col = COLUMNS.indexOf("Reference") + 1; const data = sheet.getRange(2, col, sheet.getLastRow() - 1, 1).getValues(); for (let i = 0; i < data.length; i++) { if (data[i][0] === ref) return i + 2; } return -1; } function validateToken(token) { const rowIndex = findRowByToken(token); if (rowIndex === -1) return { valid: false, reason: "Token not found" }; const sheet = getSheet(); const colMap = {}; COLUMNS.forEach((c, i) => colMap[c] = i + 1); const expiry = new Date(sheet.getRange(rowIndex, colMap["Link Expiry"]).getValue()); const used = sheet.getRange(rowIndex, colMap["Link Used"]).getValue(); const name = sheet.getRange(rowIndex, colMap["Client Name"]).getValue(); const ref = sheet.getRange(rowIndex, colMap["Reference"]).getValue(); const type = sheet.getRange(rowIndex, colMap["Client Type"]).getValue(); if (new Date() > expiry) return { valid: false, reason: "Link has expired" }; if (used === "true") return { valid: false, reason: "Link already used" }; return { valid: true, clientName: name, ref, clientType: type }; } function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // ============================================================ // 8. MANUAL HELPERS - run from the Apps Script editor // when you need to do things manually // ============================================================ // Run this to see all current FICA records in the log function listAllClients() { const sheet = getSheet(); const data = sheet.getDataRange().getValues(); data.slice(1).forEach(row => { Logger.log(row[COLUMNS.indexOf("Reference")] + " | " + row[COLUMNS.indexOf("Client Name")] + " | " + row[COLUMNS.indexOf("Status")]); }); } // Run this to manually generate a test link function testGenerateLink() { const result = generateClientLink({ clientName: "Test Client (Pty) Ltd", clientType: "Company", email: CONFIG.NOTIFY_EMAIL, phone: "+27 12 000 0000", ref: "AH/2026/TEST", }); Logger.log("Test link: " + result.intakeUrl); }
